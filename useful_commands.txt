  --entity-name amazon_reviews \
  --alter \
  --add-config min.insync.replicas=2

db.sentiment_predictions.countDocuments({
  source: "spark_structured_streaming"

db.sentiment_predictions.find(
  { source: "spark_structured_streaming" },
  {
    _id: 0,
    schema_version: 1,
    product_id: 1,
    review_date: 1,
    score: 1,
    true_label: 1,
    predicted_label: 1,
    model_type: 1,
    confidence: 1,
    confidence_available: 1,
    source_split: 1,
    batch_id: 1,
    processed_at: 1
  }
).sort({ processed_at: -1 }).limit(5).pretty()

# Sentiment distribution
db.sentiment_predictions.aggregate([
  { $match: { source: "spark_structured_streaming" } },
  { $group: { _id: "$predicted_label", count: { $sum: 1 } } },
  { $sort: { count: -1 } }
])

# Model distribution
db.sentiment_predictions.aggregate([
  { $match: { source: "spark_structured_streaming" } },
  { $group: { _id: "$model_type", count: { $sum: 1 } } },
  { $sort: { count: -1 } }
])

# ProductId required by project PDF
